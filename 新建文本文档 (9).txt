			async submitReview() {
				const item = this.reviewTarget
				const isTweet = this.adminTab.startsWith('tweet')
				const tablename = isTweet ? 'u4_Tweet' : 'u4_commit'
				const action = this.reviewAction

				try {
					if (action === 'approve') {
						await this.$request({
							action: 'update',
							tablename,
							params: { id: item.id, status: 2 }
						})
						item.status = 2
						if (this.reviewNotify) {
							this.sendNotify(item, `您的${isTweet ? '推文' : '评论'}已通过审核`, '')
						}
					} else if (action === 'reject') {
						await this.$request({
							action: 'update',
							tablename,
							params: { id: item.id, status: 0 }
						})
						item.status = 0
						if (this.reviewNotify) {
							this.sendNotify(item, `您的${isTweet ? '推文' : '评论'}审核未通过`,
								`原因: ${this.reviewReason || '未通过审核'}`)
						}
					} else if (action === 'setlevel') {
						const newLevel = Math.max(1, Math.min(9, this.reviewLevel || 1))
						await this.$request({
							action: 'update',
							tablename,
							params: { id: item.id, level: newLevel }
						})
						item.level = newLevel
						if (this.reviewNotify) {
							this.sendNotify(item, `您的${isTweet ? '推文' : '评论'}等级已调整`,
								`新等级: ${newLevel}。原因: ${this.reviewReason || '管理员调整'}`)
						}
					}

					// 如果当前tab不匹配新状态，从列表移除
					const cfg = this.adminTabConfig(this.adminTab)
					if (item.status !== cfg.status) {
						const idx = this.adminList.findIndex(t => t.id === item.id)
						if (idx !== -1) this.adminList.splice(idx, 1)
					}

					this.reviewDialog = false
					this.$youToast.success('操作成功')
				} catch (e) {
					this.$log('审核操作失败:', e)
					this.$youToast.error('操作失败')
				}
			},

			async ALLgenerateMockTweets() {
				// 动态构建 query 对象
				let query = {};

				// 只有 searchKeyword 非空时才添加 title 条件
				if (this.searchKeyword && this.searchKeyword.trim() !== '') {
					query.title = {
						$like: this.searchKeyword
					};
				}

				// 只有 searchtags 非空数组时才添加 tags 条件
				if (this.searchtags && this.searchtags.length > 0) {
					query.tags = {
						$in: this.searchtags
					};
				}

				// 合并等级权限 + status过滤
				query = this.buildQuery(query)
// this.$log(query)
				try {
					const result = this.$parseNestedJson(await this.$request({
						action: 'getList',
						tablename: 'u4_Tweet',
						params: {
							query: query,
							page: this.allpage,
							pageSize: this.allpagesize,
							 order: 'desc',
							 orderBy: 'id',
						},

					}));

					console.log('获取推文列表成功:', result);

					// 为每条推文加载媒体数据
					for (const tweet of result) {
						tweet._mediaItems = []
						if (tweet.mediald && tweet.mediald.length > 0) {
							let ids = tweet.mediald
							if (typeof ids === 'string') {
								try { ids = JSON.parse(ids) } catch (e) { ids = [] }
							}
							if (Array.isArray(ids) && ids.length > 0) {
								const mediaResult = await this.$getmediabyid(ids)
								const items = Array.isArray(mediaResult) ? mediaResult : [mediaResult]
								this.$set(tweet, '_mediaItems', items)
							}
						}
						// 解析 tags
						if (tweet.tags && typeof tweet.tags === 'string') {
							try { tweet.tags = JSON.parse(tweet.tags) } catch (e) { tweet.tags = [] }
						}
						// 解析 alist
						if (tweet.alist && typeof tweet.alist === 'string') {
							try { tweet.alist = JSON.parse(tweet.alist) } catch (e) { tweet.alist = [] }
						}
						// DB字段映射
						tweet.commentCount = tweet.comments || 0
						tweet.likes = tweet.likes || 0
						// 从本地存储读取点赞状态
						tweet.isLiked = this.likedIds.includes(tweet.id)
					}

					this.ALLtweetList = this.ALLtweetList.concat(result);
					this.ALLhasMore = result.length == this.allpagesize;
					// this.tweetList = result; // 赋值到页面状态

				} catch (error) {
					console.error('获取推文列表失败:', error.message);
					uni.showToast({
						title: '加载列表失败',
						icon: 'none',
						duration: 2000
					});
				}
			},

			// ====== 路径与界面双向绑定 ======

			// 界面 → URL
			updateUrlParams() {
				if (!this.$urlHelper) return
				this.isUpdatingFromUrl = true

				const params = { page: this.activeMenu }
				if (this.currentTweetDetail || this.currentTweetId) {
					params.tweetid = this.currentTweetId || this.currentTweetDetail.id
				} else {
					params.tweetid = null
				}
				if (this.activeMenu === 'chat' && this.chatUser) {
					params.chatuser = this.chatUser.username
					params.chatnick = this.chatUser.nickname
					params.chatcover = this.chatUser.cover
				} else {
					params.chatuser = null
					params.chatnick = null
					params.chatcover = null
				}
				if (this.activeMenu === 'author' && this.currentAuthor.username) {
					params.authoruser = this.currentAuthor.username
					params.authornick = this.currentAuthor.nickname
					params.authorcover = this.currentAuthor.cover
				} else {
					params.authoruser = null
					params.authornick = null
					params.authorcover = null
				}
				if (this.activeMenu === 'admin') {
					params.admintab = this.adminTab
				} else {
					params.admintab = null
				}

				this.$urlHelper.addParams(params)

				setTimeout(() => {
					this.isUpdatingFromUrl = false
				}, 100)
			},

			// URL → 界面
			syncFromUrlParams(params) {
				if (!params || !params.page) return
				if (!this.$urlHelper) return
				this.isUpdatingFromUrl = true
				this.currentTweetDetail = null

				this.activeMenu = params.page

				if (params.page === 'home') {
					if (this.ALLtweetList.length === 0) {
						this.ALLtweetList = []
						this.allpage = 1
						this.ALLhasMore = true
						this.loadTweets()
					}
				} else if (params.page === 'myTweets') {
					this.myTweetList = []
					this.myPage = 1
					this.myHasMore = true
					this.loadMyTweets()
				} else if (params.page === 'likes') {
					this.likedTweetList = []
					this.likedPage = 1
					this.likedHasMore = true
					this.loadLikedTweets()
				} else if (params.page === 'messages') {
					this.loadNotifications()
					this.loadChatRecords()
				} else if (params.page === 'chat' && params.chatuser) {
					// 从URL恢复聊天
					this.chatUser = {
						username: params.chatuser,
						nickname: params.chatnick || '',
						cover: params.chatcover || ''
					}
					this.chatMessages = []
					this.chatPage = 1
					this.chatHasMore = true
					this.chatInput = ''
					this.loadChatMessages()
				} else if (params.page === 'author' && params.authoruser) {
					this.currentAuthor = {
						username: params.authoruser,
						nickname: params.authornick || '',
						cover: params.authorcover || ''
					}
					this.authorWorks = []
					this.authorPage = 1
					this.authorHasMore = true
					this.loadAuthorWorks(params.authoruser)
				} else if (params.page === 'admin' && this.isAdmin) {
					this.activeMenu = 'admin'

					this.adminList = []
					this.adminPage = 1
					this.adminHasMore = true
					this.loadAdminData()
				}

						}
						if (tweet.alist && typeof tweet.alist === 'string') {
							try { tweet.alist = JSON.parse(tweet.alist) } catch (e) { tweet.alist = [] }
						}

			async loadComments() {
				try {
					const result = await this.$request({
						action: 'getList',
						tablename: 'u4_commit',
						params: {
							query: this.buildQuery({
								type: 'Tweet',
								targetid: this.currentTweetDetail.id,

							}),
							page: this.commentPage,
							pageSize: 30,
							order: 'desc',
							orderBy: 'time'
						}
					})

					let list = this.$parseNestedJson(result) || []
					// 处理评论数据
					for (const comment of list) {
						// 解析 images (JSON ID数组)
						let imgIds = []
						if (comment.images) {
							if (typeof comment.images === 'string') {
								try { imgIds = JSON.parse(comment.images) } catch (e) { imgIds = [] }
							} else if (Array.isArray(comment.images)) {
								imgIds = comment.images
							}

						comment.imageUrls = []
						if (imgIds.length > 0) {
							try {
								const mediaList = await this.$getmediabyid(imgIds)
								comment.imageUrls = mediaList.filter(m => m.Format !== 2).map(m => m.url)
							} catch (e) {
								console.error('加载评论图片失败:', e)
							}
						}
						// 解析 mention
						if (comment.mention && typeof comment.mention === 'string') {
							try { comment.mention = JSON.parse(comment.mention) } catch (e) { comment.mention = [] }
						}
						if (!Array.isArray(comment.mention)) comment.mention = []
						comment.mentionUsers = []
						// 修正字段映射 DB -> 前端
						comment.likes = comment.like || 0
						comment.soncount = comment.soncount || 0
						comment.showReplies = false
						comment.replies = []
						comment.isLiked = this.likedCommentIds.includes(comment.id)
						comment.replyPage = 1
					}

						this.detailComments = list
					} else {
						this.detailComments = this.detailComments.concat(list)
					}
					this.hasMoreComments = list.length === 30
				} catch (error) {
					console.error('加载评论失败:', error)
				}

			// 提交评论
			async submitComment() {
				if (this.guestGuard()) return
				if (!this.newComment.trim()) {
					uni.showToast({
						title: '请输入评论内容',
						icon: 'none'
					});
					return;
				}

				const isReply = this.replyTo > 0
				const parentid = isReply ? this.replyTo : 0
				const content = this.newComment.trim()

				try {
					const result = await this.$request({
						action: 'add',
						tablename: 'u4_commit',
						params: {
							type: 'Tweet',
							targetid: this.currentTweetDetail.id,
							username: this.userInfo.username,
							nickname: this.userInfo.nickname,
							cover: this.userInfo.head || '',
							content: content,
							images: JSON.stringify([]),
							Emoji: 0,
							mention: JSON.stringify([]),
							parentid: parentid,
							soncount: 0,
							status: 1,
							level: this.userInfo.grade || 0,
							like: 0,
							time: Date.now()
						}
					})

					const insertId = result && (result.insertId || (result.data && result.data.insertId))
						? (result.insertId || result.data.insertId)
						: Date.now()

					// 如果是回复，增加父评论的soncount
					if (isReply) {
						// 更新父评论的soncount
						await this.$request({
							action: 'update',
							tablename: 'u4_commit',
							params: {
								id: parentid,
								soncount: (this.replyToComment && this.replyToComment.soncount || 0) + 1
							}
						})

						// 给被回复者发送通知
						if (this.replyToComment && this.replyToComment.username !== this.userInfo.username) {
							try {
								await this.$request({
									action: 'add',
									tablename: 'u4_notify',
									params: {
										content: `${this.userInfo.nickname} 回复了你: ${content.slice(0, 50)}`,
										receivedusername: this.replyToComment.username,
										receivednickname: this.replyToComment.nickname,
										receivedcover: this.replyToComment.cover || '',
										isread: 0,
										type: 1,
										sendusername: this.userInfo.username,
										sendnickname: this.userInfo.nickname,
										sendcover: this.userInfo.head || '',
										parent: 0,
										status: 2,
										time: Date.now()
									}
								})
							} catch (e) {
								console.error('发送回复通知失败:', e)
							}
						}

						// 在父评论的replies数组中追加
						const parentComment = this.detailComments.find(c => c.id === parentid)
						if (parentComment) {
							parentComment.soncount = (parentComment.soncount || 0) + 1
							if (parentComment.showReplies && parentComment.replies) {
								parentComment.replies.push({
									id: insertId,
									_uk: 'r_' + insertId + '_' + Date.now(),
									type: 'Tweet',
									targetid: this.currentTweetDetail.id,
									username: this.userInfo.username,
									nickname: this.userInfo.nickname,
									cover: this.userInfo.head,
									content: content,
									images: [],
									imageUrls: [],
									Emoji: 0,
									mention: [],
									mentionUsers: [],
									parentid: parentid,
									soncount: 0,
									status: 2,
									level: this.userInfo.grade,
									likes: 0,
									like: 0,
									time: Date.now(),
									isLiked: false
								})
							}
						}
					} else {
						// 顶级评论，增加到列表
						this.detailComments.unshift({
							id: insertId,
							_uk: 'c_' + insertId + '_' + Date.now(),
							type: 'Tweet',
							targetid: this.currentTweetDetail.id,
							username: this.userInfo.username,
							nickname: this.userInfo.nickname,
							cover: this.userInfo.head,
							content: content,
							images: [],
							imageUrls: [],
							Emoji: 0,
							mention: [],
							mentionUsers: [],
							parentid: 0,
							soncount: 0,
							status: 2,
							level: this.userInfo.grade,
							likes: 0,
							time: Date.now(),
							isLiked: false,
							showReplies: false,
							replies: [],
							replyPage: 1
						})
						// 更新推文表评论数
						try {
							const sql = 'UPDATE u4_Tweet SET comments = comments + 1 WHERE id = ?'
							await this.$request({
								action: 'executeSql',
								tablename: 'u4_Tweet',
								params: { sql, values: [this.currentTweetDetail.id] }
							})
						} catch (e) {
							console.error('更新推文评论数失败:', e)
						}
						this.currentTweetDetail.comments = (this.currentTweetDetail.comments || 0) + 1
					}

					// 重置回复状态
					this.replyTo = 0
					this.replyToComment = null
					this.newComment = ''

					uni.showToast({
						title: isReply ? '回复成功' : '评论成功',
						icon: 'success'
					})
				} catch (error) {
					console.error('评论失败:', error)
					uni.showToast({
						title: '评论失败',
						icon: 'none'
					})
				}
			},

			likeComment(comment) {
				const newLiked = !comment.isLiked
				const id = comment.id

					if (!this.likedCommentIds.includes(id)) {
						this.likedCommentIds.push(id)

					const idx = this.likedCommentIds.indexOf(id)

				this.$storage.setItem('likedCommentIds', this.likedCommentIds)
				comment.isLiked = newLiked
				comment.likes = (comment.likes || 0) + (newLiked ? 1 : -1)
			},

			// 删除自己的评论
			async deleteComment(comment) {
				try {
					await this.$request({
						action: 'delete',
						tablename: 'u4_commit',
						params: { id: comment.id }
					})

					// 更新推文表评论数
					try {
						const sql = 'UPDATE u4_Tweet SET comments = GREATEST(0, comments - 1) WHERE id = ?'
						await this.$request({
							action: 'executeSql',
							tablename: 'u4_Tweet',
							params: { sql, values: [this.currentTweetDetail.id] }
						})
					} catch (e) {
						console.error('更新推文评论数失败:', e)
					}

					// 从列表中移除
					const idx = this.detailComments.findIndex(c => c.id === comment.id)
					if (idx !== -1) {
						this.detailComments.splice(idx, 1)
						this.currentTweetDetail.comments = Math.max(0, (this.currentTweetDetail.comments || 0) - 1)
					} else {
						// 可能是回复，从父评论的replies中移除
						for (const c of this.detailComments) {
							const rIdx = c.replies.findIndex(r => r.id === comment.id)
							if (rIdx !== -1) {
								c.replies.splice(rIdx, 1)
								c.soncount = Math.max(0, (c.soncount || 0) - 1)
								break
							}
						}
					}
					uni.showToast({ title: '删除成功', icon: 'success' })
				} catch (e) {
					console.error('删除评论失败:', e)
					uni.showToast({ title: '删除失败', icon: 'none' })
				}
			},

								islike: 1
							}

						}

						const stateResult = await this.$request({
							action: 'getList',
							tablename: 'u4_Tweet_state',
							params: {
								query: { Tweetid: id, username: this.userInfo.username },
								page: 1,
								pageSize: 1

								action: 'delete',
								tablename: 'u4_Tweet_state',
								params: { id: stateList[0].id }
							})
						}

						if (idx !== -1) {
							this.likedIds.splice(idx, 1)
						}
					}
				} catch (e) {
					console.error('同步点赞到数据库失败:', e)
				}

				// 更新推文表点赞数
				try {
					const delta = newLiked ? 1 : -1
					const sql = 'UPDATE u4_Tweet SET likes = likes + ? WHERE id = ?'
					await this.$request({
						action: 'executeSql',
						tablename: 'u4_Tweet',
						params: { sql, values: [delta, id] }
					})
				} catch (e) {
					console.error('更新推文点赞数失败:', e)
				}

				tweet.isLiked = newLiked

			// 渲染Markdown为HTML
			renderMarkdown(text) {
				if (!text) return ''
				let html = text
					// 转义HTML特殊字符
					.replace(/&/g, '&amp;')
					.replace(/</g, '&lt;')
					.replace(/>/g, '&gt;')
					// 换行转p
					.replace(/\r\n/g, '\n')
					// 代码块 ``` ``` 
					.replace(/```(\w*)\n([\s\S]*?)```/g, '<pre><code>$2</code></pre>')
					// 行内代码 `
					.replace(/`([^`]+)`/g, '<code>$1</code>')
					// 标题 ###
					.replace(/^### (.+)$/gm, '<h3>$1</h3>')
					.replace(/^## (.+)$/gm, '<h2>$1</h2>')
					.replace(/^# (.+)$/gm, '<h1>$1</h1>')
					// 加粗 ** **
					.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
					// 斜体 * *
					.replace(/\*(.+?)\*/g, '<em>$1</em>')
					// 链接 [text](url)
					.replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2" target="_blank" style="color:#4A90D9;text-decoration:underline">$1</a>')
					// 无序列表 - item
					.replace(/^- (.+)$/gm, '<li>$1</li>')
					// 数字列表
					.replace(/^\d+\.\s(.+)$/gm, '<li>$1</li>')
					// 段落（连续两个换行分段）
					.replace(/\n\n/g, '</p><p>')
					// 单换行变br
					.replace(/\n/g, '<br/>')

				html = '<p>' + html + '</p>'
				// 包裹列表
				html = html.replace(/(<li>.*?<\/li>)/g, '<ul>$1</ul>')
				html = html.replace(/<\/ul>\s*<ul>/g, '')
				// 清理空p
				html = html.replace(/<p>\s*<\/p>/g, '')
				// 处理保留的换行
				html = html.replace(/<\/p>\s*<p>/g, '</p><p>')
				return html
			},

			// 发送聊天消息
			async sendChatMessage() {
				const content = this.chatInput.trim()
				if (!content || !this.chatUser) return

				try {
					// 1. 写入通知表
					const result = await this.$request({
						action: 'add',
						tablename: 'u4_notify',
						params: {
							content: content,
							receivedusername: this.chatUser.username,
							receivednickname: this.chatUser.nickname,
							receivedcover: this.chatUser.cover,
							isread: 0,
							type: 1,
							sendusername: this.userInfo.username,
							sendnickname: this.userInfo.nickname,
							sendcover: this.userInfo.head || '',
							parent: 0,
							status: 2,
							time: Date.now()
						}
					})
					const insertId = result && (result.insertId || result.data?.insertId) || Date.now()

					// 2. 更新或创建聊天记录
					const existingResult = await this.$request({
						action: 'getList',
						tablename: 'u4_chatRecord',
						params: {
							query: {
								username: this.userInfo.username,
								sendusername: this.chatUser.username
							},
							page: 1,
							pageSize: 1
						}
					})
					const existing = this.$parseNestedJson(existingResult) || []
					if (existing.length > 0) {
						await this.$request({
							action: 'update',
							tablename: 'u4_chatRecord',
							params: {
								id: existing[0].id,
								lastcontent: content,
								time: Date.now()
							}
						})
					} else {
						await this.$request({
							action: 'add',
							tablename: 'u4_chatRecord',
							params: {
								username: this.userInfo.username,
								sendusername: this.chatUser.username,
								sendnickname: this.chatUser.nickname,
								sendcover: this.chatUser.cover,
								lastcontent: content,
								unread: 0,
								time: Date.now()
							}
						})
					}

					// 3. 给对方的聊天记录增加未读
					const theirResult = await this.$request({
						action: 'getList',
						tablename: 'u4_chatRecord',
						params: {
							query: {
								username: this.chatUser.username,
								sendusername: this.userInfo.username
							},
							page: 1,
							pageSize: 1
						}
					})
					const theirRecord = this.$parseNestedJson(theirResult) || []
					if (theirRecord.length > 0) {
						await this.$request({
							action: 'executeSql',
							tablename: 'u4_chatRecord',
							params: {
								sql: 'UPDATE u4_chatRecord SET unread = unread + 1, lastcontent = ?, time = ? WHERE id = ?',
								values: [content, Date.now(), theirRecord[0].id]
							}
						})
					} else {
						await this.$request({
							action: 'add',
							tablename: 'u4_chatRecord',
							params: {
								username: this.chatUser.username,
								sendusername: this.userInfo.username,
								sendnickname: this.userInfo.nickname,
								sendcover: this.userInfo.head || '',
								lastcontent: content,
								unread: 1,
								time: Date.now()
							}
						})
					}

					// 4. 更新对方user表的Noticenum
					try {
						const sql = 'UPDATE u4_user SET Noticenum = Noticenum + 1 WHERE username = ?'
						await this.$request({
							action: 'executeSql',
							tablename: 'u4_user',
							params: { sql, values: [this.chatUser.username] }
						})
					} catch (e) {
						console.error('更新对方通知数失败:', e)
					}

					this.chatMessages.push({
						id: insertId,
						content: content,
						receivedusername: this.chatUser.username,
						type: 1,
						sendusername: this.userInfo.username,
						sendnickname: this.userInfo.nickname,
						sendcover: this.userInfo.head || '',
						status: 2,
						time: Date.now()
					})
					this.chatInput = ''
					this.$nextTick(() => {
						this.scrollChatToBottom()
					})
				} catch (e) {
					console.error('发送消息失败:', e)
				}

				if (this.guestGuard()) return
				this.currentAuthor = {

					nickname: tweet.nickname,
					cover: tweet.cover
				}
				this.authorWorks = []
				this.authorPage = 1
				this.authorHasMore = true
				this.activeMenu = 'author'
				this.currentTweetDetail = null
				this.updateUrlParams()
				this.loadAuthorWorks(tweet.username)
			},

			// 加载作者作品
			async loadAuthorWorks(username) {
				this.authorLoading = true
				try {
					const result = await this.$request({
						action: 'getList',
						tablename: 'u4_Tweet',
						params: {
							query: this.buildQuery({ username: username }),
							page: this.authorPage,
							pageSize: 10,
							order: 'desc',
							orderBy: 'id'
						}
					})
					let list = this.$parseNestedJson(result) || []
					for (const tweet of list) {
						tweet._mediaItems = []
						if (tweet.mediald && tweet.mediald.length > 0) {
							let ids = tweet.mediald
							if (typeof ids === 'string') {
								try { ids = JSON.parse(ids) } catch (e) { ids = [] }
							}
							if (Array.isArray(ids) && ids.length > 0) {
								const mediaResult = await this.$getmediabyid(ids)
								const items = Array.isArray(mediaResult) ? mediaResult : [mediaResult]
								this.$set(tweet, '_mediaItems', items)
							}
						}
						if (tweet.tags && typeof tweet.tags === 'string') {
							try { tweet.tags = JSON.parse(tweet.tags) } catch (e) { tweet.tags = [] }
						}
						if (tweet.alist && typeof tweet.alist === 'string') {
							try { tweet.alist = JSON.parse(tweet.alist) } catch (e) { tweet.alist = [] }
						}
						tweet.commentCount = tweet.comments || 0
						tweet.likes = tweet.likes || 0
						tweet.isLiked = this.likedIds.includes(tweet.id)
					}
					if (this.authorPage === 1) {
						this.authorWorks = list
					} else {
						this.authorWorks = this.authorWorks.concat(list)
					}
					this.authorHasMore = list.length === 10
					this.authorPage++
				} catch (e) {
					console.error('加载作者作品失败:', e)
				} finally {
					this.authorLoading = false
				}
			},

			// 保存资料
			async saveProfile() {
				const updated = { ...this.editUserInfo }
				try {
					await this.$request({
						action: 'update',
						tablename: 'u4_user',
						params: {
							id: this.userInfo.id,
							email: updated.email || this.userInfo.email,
							nickname: updated.nickname || this.userInfo.nickname,
							password: updated.password || this.userInfo.password,
							head: updated.head || this.userInfo.head
						}
					})

					const oldInfo = { ...this.userInfo }

					this.userInfo = { ...this.userInfo, ...updated }
					this.editMode = false

					// 同步昵称/头像到已发布推文
					const nameChanged = updated.nickname && updated.nickname !== oldInfo.nickname
					const headChanged = updated.head && updated.head !== oldInfo.head

					if (nameChanged || headChanged) {
						await this.syncTweetProfile(nameChanged ? updated.nickname : null, headChanged ? updated.head : null)
					} else {
					}

					this.$youToast.success('保存成功')
				} catch (e) {
					this.$youToast.error('保存失败')
				}
			},

			// 同步昵称/头像到已发布推文
			async syncTweetProfile(nickname, head) {
				try {
					// 直接执行 UPDATE，不用 getCount（兼容性好）
					const setParts = []
					const values = []
					if (nickname) {
						setParts.push('nickname = ?')
						values.push(nickname)
					}
					if (head) {
						setParts.push('cover = ?')
						values.push(head)
					}
					values.push(this.userInfo.username)
					const sql = `UPDATE u4_Tweet SET ${setParts.join(', ')} WHERE username = ?`

					await this.$request({
						action: 'executeSql',
						tablename: 'u4_Tweet',
						params: { sql, values }
					})

					// 查一下影响了多少条（确认结果）
					const countResult = this.$parseNestedJson(await this.$request({
						action: 'getCount',
						tablename: 'u4_Tweet',
						params: { query: { username: this.userInfo.username } }
					}))
					const total = typeof countResult === 'number' ? countResult : (countResult && countResult.count) || 0
					

					if (total > 0) {
						this.showProgress(total)
						this.updateProgress(total, `已同步 ${total} 条推文`)
						setTimeout(() => { this.hideProgress() }, 800)
					}
				} catch (e) {
					this.hideProgress()
					this.$youToast.error('同步推文信息失败')
				}
			},

			async logout() {
				try {
					if(this.$storage.getItem('isGuest')){
						this.$storage.setItem('isGuest',false);
						uni.showToast({ title: '退出成功', icon: 'success' })
						setTimeout(() => {
							uni.navigateTo({ url: '/pages/index/index2' })
						}, 800)
						return
					}

					const currentDeviceId = getDeviceId();
					let deviceList = Array.isArray(this.userInfo.deviceList) ? this.userInfo.deviceList : [];
					deviceList = deviceList.filter(item => item.id !== currentDeviceId);

					this.$set(this, 'userInfo', {
						...this.userInfo,
						deviceList: deviceList
					});

					await this.$request({
						action: 'update',
						tablename: 'u4_user',
						params: {
							id: this.userInfo.id,
							deviceList: deviceList
						}
					});

					if (this.$storage && typeof this.$storage.removeItem === 'function') {
						this.$storage.removeItem('currentUser');
					}

					uni.showToast({ title: '退出成功', icon: 'success' })
					setTimeout(() => {
						uni.navigateTo({ url: '/pages/index/index2' })
					}, 800)
				} catch (error) {
					console.error('退出登录失败:', error);
					uni.showToast({ title: '退出失败', icon: 'none' })
				}
			},

			isBilibiliVideo(url) {

			},

			// 格式化媒体URL（处理协议相对地址）
			getMediaUrl(url) {
				if (!url) return ''

				return url
			},

			// 获取B站嵌入播放器地址
			getBilibiliEmbedUrl(url) {
				const bvMatch = url.match(/BV[a-zA-Z0-9]+/)
				if (!bvMatch) return url
				const bvid = bvMatch[0]
				return `https://www.bilibili.com/blackboard/html5mobileplayer.html?bvid=${bvid}&autoplay=0&muted=1&high_quality=1&danmaku=0`
			},

			},

			// 格式化时间
			formatTime(timestamp) {
				const now = Date.now();
				const diff = now - timestamp;

				const minute = 60 * 1000;
				const hour = 60 * minute;
				const day = 24 * hour;
				const month = 30 * day;

				if (diff < minute) {
					return '刚刚';
				} else if (diff < hour) {
					return Math.floor(diff / minute) + '分钟前';
				} else if (diff < day) {
					return Math.floor(diff / hour) + '小时前';
				} else if (diff < month) {
					return Math.floor(diff / day) + '天前';
				} else {
					const date = new Date(timestamp);
					return `${date.getFullYear()}-${date.getMonth() + 1}-${date.getDate()}`;
				}
			}
		}
